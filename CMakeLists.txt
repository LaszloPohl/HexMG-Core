cmake_minimum_required(VERSION 3.24)
project(HexMGCoreOptimized LANGUAGES C CXX)
add_library(hexmg_binary_codec STATIC third_party/miniz/miniz.c third_party/miniz/miniz_tinfl.c third_party/miniz/miniz_tdef.c)
set_target_properties(hexmg_binary_codec PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
target_include_directories(hexmg_binary_codec PUBLIC third_party/miniz)
target_compile_definitions(hexmg_binary_codec PUBLIC MINIZ_NO_ARCHIVE_APIS MINIZ_NO_STDIO MINIZ_NO_ZLIB_COMPATIBLE_NAMES)
include(CTest)
add_library(hexmg_core STATIC src/hmgComponent.cpp src/hmgComponentCreate.cpp
    src/hmgHMGFileReader.cpp src/hmgSimulation.cpp src/hmgSunred.cpp)
target_include_directories(hexmg_core PUBLIC src)
target_link_libraries(hexmg_core PUBLIC hexmg_binary_codec)
target_compile_features(hexmg_core PUBLIC cxx_std_20)
add_executable(hexmg_core_optimized src/hmgMain.cpp)
target_link_libraries(hexmg_core_optimized PRIVATE hexmg_core)
if(BUILD_TESTING)
    add_executable(function_derivative_tests tests/function_derivative_tests.cpp)
    target_link_libraries(function_derivative_tests PRIVATE hexmg_core)
    set_target_properties(function_derivative_tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    add_executable(gmd_jacobian_tests tests/gmd_jacobian_tests.cpp)
    target_link_libraries(gmd_jacobian_tests PRIVATE hexmg_core)
    set_target_properties(gmd_jacobian_tests PROPERTIES AUTOMOC OFF AUTOUIC OFF AUTORCC OFF)
    add_executable(convergence_metric_tests tests/convergence_metric_tests.cpp)
    target_include_directories(convergence_metric_tests PRIVATE src)
    target_compile_features(convergence_metric_tests PRIVATE cxx_std_20)
    add_test(NAME convergence_metric COMMAND convergence_metric_tests)
    add_executable(optional_reader_tests tests/optional_reader_tests.cpp)
    target_link_libraries(optional_reader_tests PRIVATE hexmg_core)
    add_test(NAME optional_reader COMMAND ${CMAKE_COMMAND}
        "-DREADER=$<TARGET_FILE:optional_reader_tests>"
        "-DFIXTURE_ROOT=${CMAKE_CURRENT_BINARY_DIR}/optional-fixtures"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/tests/run_optional_reader.cmake")
endif()
foreach(target hexmg_core hexmg_core_optimized optional_reader_tests convergence_metric_tests gmd_jacobian_tests function_derivative_tests)
    if(TARGET ${target} AND MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /Zc:__cplusplus)
    endif()
endforeach()
